cmake_minimum_required(VERSION 3.24)
project(mixtape VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(MSVC)
  # Static CRT so the .exe is a single portable file.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8 /W3 /MP)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

include(FetchContent)
set(FETCHCONTENT_QUIET OFF)
# Removable / exFAT drives have no ownership info, which makes git refuse to touch
# the fetched repositories ("dubious ownership"). Scope the override to this build.
set(ENV{GIT_CONFIG_COUNT} 1)
set(ENV{GIT_CONFIG_KEY_0} safe.directory)
set(ENV{GIT_CONFIG_VALUE_0} *)

# ---- GLFW -------------------------------------------------------------------
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG 3.4
  GIT_SHALLOW TRUE)

# ---- Dear ImGui (docking branch) -------------------------------------------
FetchContent_Declare(imgui
  GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG v1.91.9b-docking
  GIT_SHALLOW TRUE)

# ---- TagLib (audio metadata) -----------------------------------------------
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
FetchContent_Declare(taglib
  GIT_REPOSITORY https://github.com/taglib/taglib.git
  GIT_TAG v2.0.2
  GIT_SHALLOW TRUE)

# ---- nlohmann/json ----------------------------------------------------------
set(JSON_BuildTests OFF CACHE INTERNAL "")
set(JSON_Install OFF CACHE INTERNAL "")
FetchContent_Declare(json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG v3.11.3
  GIT_SHALLOW TRUE)

# ---- JetBrains Mono (embedded UI font) --------------------------------------
FetchContent_Declare(jbmono
  URL https://github.com/JetBrains/JetBrainsMono/releases/download/v2.304/JetBrainsMono-2.304.zip
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)

FetchContent_MakeAvailable(glfw imgui taglib json jbmono)

# ---- ImGui static lib -------------------------------------------------------
add_library(imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/imgui_demo.cpp
  ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC
  ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${imgui_SOURCE_DIR}/misc/cpp)
target_compile_definitions(imgui PUBLIC IMGUI_DEFINE_MATH_OPERATORS)
find_package(OpenGL REQUIRED)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)

# ---- Embed fonts ------------------------------------------------------------
include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})
embed_file(${jbmono_SOURCE_DIR}/fonts/ttf/JetBrainsMono-Regular.ttf ${GEN_DIR}/font_regular.h g_font_regular)
embed_file(${jbmono_SOURCE_DIR}/fonts/ttf/JetBrainsMono-Bold.ttf    ${GEN_DIR}/font_bold.h    g_font_bold)
# window icon (raw RGBA, generated by assets/make_icon.py)
embed_file(${CMAKE_CURRENT_SOURCE_DIR}/assets/icon-64.rgba ${GEN_DIR}/icon64.h g_icon64)
embed_file(${CMAKE_CURRENT_SOURCE_DIR}/assets/icon-32.rgba ${GEN_DIR}/icon32.h g_icon32)

# ---- mixtape ----------------------------------------------------------------
add_executable(mixtape
  src/main.cpp
  src/app.cpp
  src/library.cpp
  src/playlist.cpp
  src/platform.cpp
  src/theme.cpp
  src/util.cpp
  src/filebrowser.cpp)
target_include_directories(mixtape PRIVATE
  ${GEN_DIR}
  ${taglib_SOURCE_DIR}/taglib
  ${taglib_SOURCE_DIR}/taglib/toolkit
  ${taglib_BINARY_DIR})
target_compile_definitions(mixtape PRIVATE TAGLIB_STATIC)
target_link_libraries(mixtape PRIVATE imgui tag nlohmann_json::nlohmann_json)

if(WIN32)
  target_sources(mixtape PRIVATE assets/mixtape.rc)  # taskbar / explorer icon
  target_link_libraries(mixtape PRIVATE dwmapi ole32 shell32 uuid)
  if(MSVC)
    # GUI subsystem, but keep a normal main()
    target_link_options(mixtape PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
  else()
    target_link_options(mixtape PRIVATE -mwindows)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(mixtape PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()

install(TARGETS mixtape RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES assets/mixtape.desktop DESTINATION share/applications)
  foreach(sz 16 32 48 64 128 256 512)
    install(FILES assets/icon-${sz}.png DESTINATION share/icons/hicolor/${sz}x${sz}/apps RENAME mixtape.png)
  endforeach()
  install(FILES assets/mixtape.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()

# ---- tests ------------------------------------------------------------------
option(MIXTAPE_BUILD_TESTS "Build the playlist round-trip test" ON)
if(MIXTAPE_BUILD_TESTS)
  enable_testing()
  add_executable(playlist_test tests/playlist_test.cpp src/playlist.cpp src/util.cpp)
  target_include_directories(playlist_test PRIVATE src)
  add_test(NAME playlist_roundtrip COMMAND playlist_test)
endif()
